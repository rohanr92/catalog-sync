/* ===== EXTEND BACKGROUND =====
   Continues the photo's own studio background out to the canvas edges.
   The photo pixels are never changed; only the empty area around it is filled. */
function _xrng(seed){let a=(seed>>>0)||1;return()=>{a=(a+0x6D2B79F5)|0;let t=Math.imul(a^(a>>>15),1|a);t=(t+Math.imul(t^(t>>>7),61|t))^t;return((t^(t>>>14))>>>0)/4294967296;};}
function _xg(r){return (r()+r()+r()-1.5)*2;}
function _xmed(a){a.sort((p,q)=>p-q);return a.length?a[a.length>>1]:0;}
function placeForExtend(rec,where){
  const rot=((rec.transform.rot||0)%360+360)%360, swap=rot===90||rot===270;
  const iw=swap?rec.h:rec.w, ih=swap?rec.w:rec.h;
  const s=Math.min(state.canvasW/iw,state.canvasH/ih);
  rec.transform.scale=s;
  const exW=state.canvasW-iw*s, exH=state.canvasH-ih*s;
  let x=0,y=0;
  if(exW>=exH){ if(where==='left')x=exW/2; else if(where==='right')x=-exW/2; }
  else { if(where==='left')y=exH/2; else if(where==='right')y=-exH/2; }
  rec.transform.x=x; rec.transform.y=y;
  rec.bg.mode='extend'; rec.bg.place=where; if(rec.bg.blend==null) rec.bg.blend=50;
}
function extendFill(c,rec,W,H,cx,cy,dw,dh,swap){
  const iw=swap?dh:dw, ih=swap?dw:dh;
  const r={x0:Math.max(0,Math.ceil(cx-iw/2)),y0:Math.max(0,Math.ceil(cy-ih/2)),x1:Math.min(W,Math.floor(cx+iw/2)),y1:Math.min(H,Math.floor(cy+ih/2))};
  if(r.x1-r.x0<6||r.y1-r.y0<6) return;
  let id; try{ id=c.getImageData(0,0,W,H); }catch(e){ return; }
  const d=id.data, blend=Math.max(0,Math.min(1,(rec.bg.blend==null?50:rec.bg.blend)/100)), rnd=_xrng(rec.id*7919+W*31+H);
  let busy=_extAxis(d,W,H,r,true,blend,rnd);
  busy=Math.max(busy,_extAxis(d,W,H,{x0:0,y0:r.y0,x1:W,y1:r.y1},false,blend,rnd));
  _fillHoles(d,W,H,r,rnd);
  c.putImageData(id,0,0);
  rec._extBusy=busy;
}
function _extAxis(d,W,H,r,hor,blend,rnd){
  const L=hor?r.y1-r.y0:r.x1-r.x0, off=hor?r.y0:r.x0;

      let s=0;
      [t0,t1,t2].forEach((a,ch)=>{let m=0;for(const v of a)m+=v;m/=a.length;let q=0;for(const v of a)q+=(v-m)*(v-m);s+=Math.sqrt(q/a.length);med[l*3+ch]=_xmed(a);});

    }
    // rows where the model/garment reaches the photo edge: don't smear them outward
    const win=Math.max(5,Math.round(L*0.2))|1, h2=win>>1, st=Math.max(1,Math.floor(win/60));
    const glob=[0,1,2].map(ch=>{const a=[];for(let l=0;l<L;l++)a.push(med[l*3+ch]);return _xmed(a);});
    const intr=new Uint8Array(L);

      }
      if(dev>14||far>45||sd[l]>14) intr[l]=1;
    }
    const g=Math.max(2,Math.round(L/400)), bad=new Uint8Array(L);
    for(let l=0;l<L;l++) if(intr[l]) for(let j=Math.max(0,l-g);j<=Math.min(L-1,l+g);j++) bad[j]=1;
    let nb=0; for(let l=0;l<L;l++) nb+=bad[l]; busyMax=Math.max(busyMax,nb/L);
    if(nb>0&&nb<L){
      let prev=-1;
      for(let l=0;l<=L;l++){
        if(l<L&&bad[l]) continue;
        for(let j=prev+1;j<l;j++) for(let ch=0;ch<3;ch++){
          const a=prev>=0?med[prev*3+ch]:med[l*3+ch], b=l<L?med[l*3+ch]:a, t=(prev>=0&&l<L)?(j-prev)/(l-prev):0;
          med[j*3+ch]=a+(b-a)*t;
        }
        prev=l;
      }
    }
    const sig=Math.max(1.5,L*0.004), R=Math.ceil(sig*3), ker=[]; let ks=0;
    for(let j=-R;j<=R;j++){const v=Math.exp(-j*j/(2*sig*sig));ker.push(v);ks+=v;}
    const sm=new Float32Array(L*3);
    for(let l=0;l<L;l++) for(let ch=0;ch<3;ch++){let s=0;for(let j=-R;j<=R;j++){const jj=Math.min(L-1,Math.max(0,l+j));s+=med[jj*3+ch]*ker[j+R];}sm[l*3+ch]=s/ks;}
    const res=[], rs=Math.max(1,Math.floor(L/300));
    for(let l=0;l<L;l+=rs){ if(bad[l]) continue; for(let p=sStart;p<sEnd;p++){const i=P(l,p);res.push(Math.abs(d[i+1]-sm[l*3+1]));} }
    const grain=Math.min(6,Math.max(0.3,_xmed(res)*1.4826));
    const f=Math.max(1,Math.min(f0,depth,span-2));
    for(let l=0;l<L;l++){
      const calm=bad[l]?0:Math.max(0,Math.min(1,1-(sd[l]-6)/20)), wm=calm*Math.min(1,blend*2);
      for(let q=0;q<depth;q++){
        const pos=low?a0-1-q:a1+q, i=P(l,pos), n=_xg(rnd)*grain;

      }
    }
  }
  return busyMax;
}
/* anything erased inside the photo (magic eraser) is filled with the surrounding background */
function _boxAvg(src,w,h,rad,out){
  const ii=new Float64Array((w+1)*(h+1));
  for(let y=0;y<h;y++){let row=0;for(let x=0;x<w;x++){row+=src[y*w+x];ii[(y+1)*(w+1)+x+1]=ii[y*(w+1)+x+1]+row;}}
  for(let y=0;y<h;y++){const ya=Math.max(0,y-rad),yb=Math.min(h,y+rad+1);
    for(let x=0;x<w;x++){const xa=Math.max(0,x-rad),xb=Math.min(w,x+rad+1);
      out[y*w+x]=(ii[yb*(w+1)+xb]-ii[ya*(w+1)+xb]-ii[yb*(w+1)+xa]+ii[ya*(w+1)+xa])/((yb-ya)*(xb-xa));}}
}
function _fillHoles(d,W,H,r,rnd){

  if(x1<0) return;
  if(cnt>(r.x1-r.x0)*(r.y1-r.y0)*0.25) return;   // background was removed on purpose - leave it

  const bw=x1-x0+1, bh=y1-y0+1, n=bw*bh, mk=new Uint8Array(n), ch=[new Float32Array(n),new Float32Array(n),new Float32Array(n)], tmp=new Float32Array(n);
  for(let y=0;y<bh;y++){ let s=[0,0,0],c2=0;
    for(let x=0;x<bw;x++){const i=((y0+y)*W+x0+x)*4,k=y*bw+x; for(let c=0;c<3;c++)ch[c][k]=d[i+c]; if(d[i+3]<250)mk[k]=1; else {s[0]+=d[i];s[1]+=d[i+1];s[2]+=d[i+2];c2++;}}
    for(let x=0;x<bw;x++){const k=y*bw+x; if(mk[k]&&c2) for(let c=0;c<3;c++) ch[c][k]=s[c]/c2;} }
  const big=Math.max(3,Math.round(Math.min(bw,bh)/12));
  for(const [rad,it] of [[big,5],[Math.max(2,big>>2),8],[1,10]]) for(let t=0;t<it;t++) for(let c=0;c<3;c++){ _boxAvg(ch[c],bw,bh,rad,tmp); for(let k=0;k<n;k++) if(mk[k]) ch[c][k]=tmp[k]; }

}
